cmake_minimum_required(VERSION 3.18)
project(hbr_core CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

add_library(hbr_core STATIC src/hbr_core.cpp)
target_include_directories(hbr_core PUBLIC include)
set_target_properties(hbr_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
find_package(Threads)
if(Threads_FOUND)
  target_link_libraries(hbr_core PUBLIC Threads::Threads)
endif()

# Shared library for the desktop validation harness (tools/harness).
option(HBR_BUILD_SHARED "Build libhbr_core shared library" OFF)
if(HBR_BUILD_SHARED)
  add_library(hbr_core_shared SHARED src/hbr_core.cpp)
  target_include_directories(hbr_core_shared PUBLIC include)
  set_target_properties(hbr_core_shared PROPERTIES OUTPUT_NAME hbr_core)
  if(Threads_FOUND)
    target_link_libraries(hbr_core_shared PUBLIC Threads::Threads)
  endif()
endif()

option(HBR_BUILD_TESTS "Build unit tests" OFF)
if(HBR_BUILD_TESTS)
  enable_testing()
  add_executable(hbr_core_tests tests/test_hbr_core.cpp)
  target_link_libraries(hbr_core_tests hbr_core)
  add_test(NAME hbr_core_tests COMMAND hbr_core_tests)
endif()
